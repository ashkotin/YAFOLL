НО при этом КСГ часть правила не должна начинаться с \t, т.е. отступ пробелами.

/*Логика действий
1. основная идея - действия лишь(!) строят ДРВ.
2. но ДРВ - EBNF, в котором есть постфиксы "+" и "*" (см. комментарий с EBNF). поэтому нам нужна трансляция этих операторов в БНФ. 
При этом "конфликтность" bison привела к необходимости уже для + прибегнуть к двух вариантам трансляции. 
что усложнило (обобщило;-) нумерацию членов +-правила (иногда они имеют отрицательные номера!).
*/

%glr-parser
/*каждая строка token соответствует(?) строке РВ, а %token нескольким!*/
%token EXISTS
%token FOR_ANY
%token Dot
%token COMMA
%token COLON
%token l_p
%token r_p
%token e_m
%token q_m

%start Statements

%%
/*<EBNF>Statements:Statement+.*/
Statements :  Statement
   | Statements Statement
;
Statement : String /*Хранимый комментарий:-)*/
   | Declaration
   | Proof Id Id derived_formula Dot 
   | e_m Ide Id e_m 
   | e_m Ide e_m 
   | e_m term COLON e_m 
   | e_m term COLON term e_m   /*КМАС. Задание значения терма-1 равным term-2.*/
   | q_m term q_m 
   | e_m Number e_m   
   | e_m error e_m
   | error Dot
;

/*Объявления*/
Declaration : DECLARATION Id sort Dot
   | DECLARATION Id sort String Dot
   | DECLARATION Id sig_f PRIME Dot
   | DECLARATION Id sig_f PRIME Id Dot
   | DECLARATION Id sig_f DEFINITION Id_list_bch COLON term Dot
;

/***Выведенная формула***/
derived_formula : LA1 l_p term COMMA term r_p 
   | LA2 l_p term COMMA term COMMA term r_p 
   | LA3 l_p term COMMA term r_p 
   | LA4 l_p Id COMMA term COMMA term r_p 
   | LA5 l_p Id COMMA term COMMA term r_p 
   | MP l_p derived_formula COMMA derived_formula r_p 
   | Gen l_p Id COMMA derived_formula r_p 
   | Id 
;
term : Id 
   /*(_trmRE) term:Ide|Number|String|Year. при обработке терма может встретиться правило _trmRE, единственный ребё которого - лнт!*/
term : Ide 
term : Number
term : String
term : Year
term : l_p Id term r_p 
term : l_p term Id term r_p
term : term l_p TermList r_p
term : l_p EXISTS  Id COLON Id term r_p
term : l_p FOR_ANY Id COLON Id term r_p
/*<EBNF>(#trml)

TermList : 
   | TermList_p 
; 
/*<EBNF>(#trml)
TermList_p : term
 /*Z:Перестановка ведёт к: 2 конфликта сдвига/вывода!!! Возможно с Id_list либо правилами вывода и можно вводить ","*/
   | term TermList_p 
; 

/*Для форм- параметров определений.*/
/*<EBNF> Id_list : Id* */
Id_list :  
 | Id_listp  
;
/*<EBNF> Id_listp : Id+ */
Id_listp : Id 
 | Id_listp Id  
;
Id_list_b : l_p Id_list r_p 
;
/*<EBNF> Id_list_bch : Id_list_b+*/
Id_list_bch : Id_list_b 
 | Id_list_bch Id_list_b 
;
/*Синтаксис сигнатур*/
sig_f : sig_arg COLON sig_e 
;
sig_e : Id
   | l_p sig_f r_p 
;
/*<EBNF> sig_arg : sig_e* */
sig_arg :  
   |  sig_ep 
;
/*<EBNF> sig_ep : sig_e+ *//*!!!будучи нацелена на sig_arg* эта группа правил снабжена соответствующими действиями построения частей *-дерева!!!*/
sig_ep : sig_e 
   |sig_ep  sig_e
;
